=org-babel-load-file=, which tangles all the =emacs-lisp= source blocks below
into =config.el= and evaluates them.

* Lexical binding

The tangled =config.el= needs a =lexical-binding= cookie on its very first
line.  Org's =:lexical t= header only controls in-buffer evaluation, not the
tangled output, so without this the file loads under dynamic binding and
closures (e.g. the dashboard buttons) break.

* Dashboard

A simple startup dashboard with pinned paths and bookmarks.

#+begin_src emacs-lisp
(defvar my/pinned-paths
  '(
    ("Config"     . "~/.emacs.d/config.org")
    ("OneDrive-Work"         . " ~/OneDrive/1000-Work/"))
   )

(defun my/insert-file-button (label path)
  (insert-button label
                 'action (lambda (_btn) (find-file path))
                 'follow-link t
                 'face 'link)
  (insert "\n"))

(defun my/dashboard-refresh ()
  (interactive)
  (let ((buf (get-buffer-create "*dashboard*")))
    (with-current-buffer buf
      (let ((inhibit-read-only t))
        (erase-buffer)
        (insert (propertize "Emacs\n\n" 'face '(:height 1.5 :weight bold)))

        (when my/pinned-paths
          (insert (propertize "Pinned\n" 'face 'bold))
          (dolist (p my/pinned-paths)
            (my/insert-file-button (car p) (cdr p))))

        (insert (propertize "\nBookmarks\n" 'face 'bold))
        (require 'bookmark)
        (bookmark-maybe-load-default-file)
        (if (bookmark-all-names)
            (dolist (bm (bookmark-all-names))
              (insert-button bm
                             'action (lambda (_btn) (bookmark-jump bm))
                             'follow-link t
                             'face 'link)
              (insert "\n"))
          (insert (propertize "  (none yet — C-x r m to add one)\n" 'face 'shadow)))

        (goto-char (point-min))
        (special-mode)
        (local-set-key (kbd "g") #'my/dashboard-refresh)))
    buf))

(setq initial-buffer-choice #'my/dashboard-refresh)
#+end_src

;; C-c C-v: open the PDF built from the current LaTeX buffer.
(defun my/latex-view-pdf ()
  "Open the PDF built from the current LaTeX buffer."
  (interactive)
  (let ((pdf (concat (file-name-sans-extension buffer-file-name) ".pdf")))
    (if (file-exists-p pdf)
        (find-file-other-window pdf)
      (message "No PDF yet: %s" pdf))))

(with-eval-after-load 'tex-mode
  (define-key latex-mode-map (kbd "C-c C-v") #'my/latex-view-pdf))

;; C-c C-g: forward SyncTeX search — jump from the current .tex line to the
;; matching spot in the PDF (pdf-tools).  pdf-tools' own `pdf-sync-forward-search'
;; requires AUCTeX, so this calls the poppler backend directly with base.pdf.
(defun my/latex-forward-search ()
  "Jump from the current LaTeX line to the corresponding place in the PDF."
  (interactive)
  (require 'pdf-tools)
  (let* ((tex (buffer-file-name))
         (pdf (concat (file-name-sans-extension tex) ".pdf"))
         (line (line-number-at-pos nil t))
         (column (current-column)))
    (unless (file-exists-p pdf)
      (user-error "No PDF yet: %s (compile first)" pdf))
    (let-alist (pdf-info-synctex-forward-search tex line column pdf)
      (let ((buffer (or (find-buffer-visiting pdf)
                        (find-file-noselect pdf))))
        (with-selected-window (display-buffer buffer)
          (pdf-util-assert-pdf-window)
          (when .page
            (pdf-view-goto-page .page (selected-window))
            (let ((y1 (nth 1 .edges)))
              (when y1
                (pdf-util-tooltip-arrow
                 (round (* y1 (cdr (pdf-view-image-size)))))))))))))

(with-eval-after-load 'tex-mode
  (define-key latex-mode-map (kbd "C-c C-g") #'my/latex-forward-search))
#+end_src

* Sync-scroll

Keep windows that show the same file (a buffer and its indirect clones)
aligned by BUFFER POSITION.  Unlike =scroll-all-mode= this follows
mouse/trackpad too, and aligns by content even when one window is taller
(rendered equation images).  Toggle with =C-c s=.

#+begin_src emacs-lisp
(defvar my/sync-scroll--busy nil)

(defun my/sync-scroll--run (win start)
  "Align other windows on the same underlying file to START."
  (when (and my/sync-scroll-mode (not my/sync-scroll--busy))
    (let* ((my/sync-scroll--busy t)
           (b (window-buffer win))
           (root (or (buffer-base-buffer b) b)))
      (dolist (w (window-list nil 'no-mini))
        (unless (eq w win)
          (let* ((wb (window-buffer w))
                 (wroot (or (buffer-base-buffer wb) wb)))
            (when (eq wroot root)               ; same file text
              (set-window-start w start t)))))))) ; t = don't force, no error

(define-minor-mode my/sync-scroll-mode
  "Globally keep windows showing the same file aligned by position."
  :global t
  (if my/sync-scroll-mode
      (add-hook 'window-scroll-functions #'my/sync-scroll--run)
    (remove-hook 'window-scroll-functions #'my/sync-scroll--run)))

(global-set-key (kbd "C-c s") #'my/sync-scroll-mode)
#+end_src
